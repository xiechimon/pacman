import { expect, type Page, test } from '@playwright/test';
import { evidenceShot } from './evidence';

// BEFORE-evidence probe for #728 (runs on origin/main ONLY — it pins the
// OLD behavior and must fail on the fixed branch). One test per defect the
// ticket recon named, each capturing the observable state.

const TEAM_ID = 'team-1';
const USER = { id: 'user-1', displayName: '我', avatarUrl: null };
const TEAM = { id: TEAM_ID, name: 'Team', createdAt: 0, plan: 'free', avatarStyle: null };
const CARD_ID = 'todo-1';
const BUILD_ID = 'build-1';
const PROJECT_ID = 'proj-1';
const WIRE_CARD = { id: CARD_ID, teamId: TEAM_ID, projectId: PROJECT_ID, title: 'BEFORE 探针', spec: '', phase: 'building', phaseAt: 0, seqNum: 9, orderIndex: 0, tagIds: [], assignment: null, agent: null, latestBuildId: BUILD_ID, lastRunAt: 0, hasChanges: false, hasPlan: true, buildHistory: [], sourceTodo: null, v: 1 };
const PROJECT = { id: PROJECT_ID, name: 'pacman', teamId: TEAM_ID, repoKind: 'hosted', repoName: 'pacman', githubRepo: null, localPath: null };
const WIRE_BUILD = { id: BUILD_ID, todoId: CARD_ID, withPlan: true, prevPhase: null, triggerSource: 'user', pinnedMachineId: null, planDocId: null, errorMessage: null, prUrl: null, prNumber: null, diffHash: null, createdAt: 0 };
const EMPTY_CONVERSATION = { messages: [], chips: null, historyEpoch: 0, steerPending: [], activeRun: null, nextCursor: null };
const AGENTS = [
  { id: 'agent-1', displayName: 'builder', description: 'Builds features' },
  { id: 'agent-2', displayName: 'reviewer', description: 'Reviews PRs' },
  { id: 'agent-3', displayName: 'deploy-bot', description: 'Ships to prod' },
];

async function openDetail(page: Page, postStatus = 201) {
  await page.route('**/api/**', (route, request) => {
    if (request.method() !== 'GET') return route.fallback();
    return route.fulfill({ status: 500, json: { error: 'stub' } });
  });
  await page.route('**/api/teams', (route) => route.fulfill({ json: [TEAM] }));
  await page.route('**/api/user/me', (route) => route.fulfill({ json: USER }));
  await page.route(`**/api/teams/${TEAM_ID}/notifications`, (route) => route.fulfill({ json: { unreadThreadIds: [] } }));
  const members = [
    { id: 'member-user', teamId: TEAM_ID, actorId: USER.id, memberType: 'user', actor: USER },
    ...AGENTS.map((a, i) => ({ id: `member-agent-${i}`, teamId: TEAM_ID, actorId: a.id, memberType: 'agent', actor: a })),
  ];
  await page.route(`**/api/teams/${TEAM_ID}/members`, (route) => route.fulfill({ json: members }));
  await page.route('**/api/todos?*', (route) => route.fulfill({ json: [WIRE_CARD] }));
  await page.route(`**/api/todos/${CARD_ID}`, (route) => route.fulfill({ json: WIRE_CARD }));
  await page.route('**/api/projects', (route) => route.fulfill({ json: [PROJECT] }));
  await page.route(`**/api/builds/${BUILD_ID}`, (route) => route.fulfill({ json: WIRE_BUILD }));
  await page.route(`**/api/builds/${BUILD_ID}/steps`, (route) => route.fulfill({ json: [] }));
  await page.route(`**/api/builds/${BUILD_ID}/plans`, (route) => route.fulfill({ json: [] }));
  await page.route(`**/api/builds/${BUILD_ID}/usage`, (route) => route.fulfill({ json: [] }));
  await page.route(`**/api/builds/${BUILD_ID}/changes`, (route) => route.fulfill({ json: { files: [] } }));
  const sent: string[] = [];
  await page.route(`**/api/conversations/${BUILD_ID}/messages`, (route, request) => {
    if (request.method() !== 'POST') return route.fulfill({ json: EMPTY_CONVERSATION });
    sent.push(request.postData() ?? '');
    return route.fulfill({ status: postStatus, json: postStatus === 201 ? { message: { id: 'msg-1' } } : { error: 'no active step' } });
  });
  const membersLoaded = page.waitForResponse(
    (r) => r.url().includes(`/api/teams/${TEAM_ID}/members`) && r.status() === 200,
  );
  await page.goto(`/app/todo/${CARD_ID}`);
  // The inline list only arms once the roster is in (main re-judges only on
  // change events) — type before the members resolve and the popup never
  // opens for the already-typed token.
  await membersLoaded;
  const input = page.locator('.composer-input');
  await expect(input).toBeEditable();
  return { input, listbox: page.locator('.mention-inline'), rows: page.locator('.mention-inline-row'), sent };
}

test('BEFORE: mid-word @ opens the listbox (no left-boundary check)', async ({ page }) => {
  const { input, listbox } = await openDetail(page);
  await input.fill('foo@');
  // DEFECT: rules 2-3 say this must NOT trigger.
  await expect(listbox).toBeVisible();
  await evidenceShot(page, 'before-midword-trigger.png');
});

test('BEFORE: Enter with the listbox open SENDS the message (r9 §5 conflict)', async ({ page }) => {
  const { input, listbox, rows, sent } = await openDetail(page);
  await input.fill('@bu');
  await expect(listbox).toBeVisible();
  await evidenceShot(page, 'before-enter-listbox-open.png');
  // On main, Enter's outcome RACES the focus-steal effect (r9 §5 logged
  // exactly this: "单次观测疑似焦点竞态"). Prime the textarea facet so this
  // probe pins it deterministically: with the textarea focused, Enter sends
  // the message even though the listbox is open.
  await input.evaluate((el) => el.focus());
  await input.press('Enter');
  // DEFECT: the message left the composer instead of accepting a row.
  await expect.poll(() => sent.length).toBe(1);
  expect(sent[0]).toContain('@bu');
  await expect(input).toHaveValue('');
  // DEFECT (worse than the ticket text): the send does not close the
  // listbox either — the stale popup floats over the now-empty composer.
  await expect(listbox).toBeVisible();
  await expect(rows).toHaveCount(3);
  await evidenceShot(page, 'before-enter-sent.png');
});

test('BEFORE: Enter on the stolen-focus row inserts the LAST row with residue (race facet B)', async ({ page }) => {
  const { input, rows } = await openDetail(page);
  await input.fill('@');
  await expect(rows).toHaveCount(3);
  // The focus-steal effect lands on the LAST row (shared ref defect), so
  // Enter activates deploy-bot — not what the user was reading — and the
  // insert leaves the `@` residue.
  // Wait for the steal to land (it is an effect — timing varies), then pin
  // the row facet: Enter activates the LAST row through the shared-ref defect.
  await page.waitForFunction(() =>
    document.activeElement?.classList?.contains('mention-inline-row'),
  );
  const focused = await input.evaluate(() => document.activeElement?.textContent ?? '');
  expect(focused).toContain('deploy-bot');
  // page.keyboard (not locator.press — that would refocus the textarea and
  // test facet A again): the raw Enter lands on the focused row button and
  // activates it as a click.
  await page.keyboard.press('Enter');
  await expect(input).toHaveValue('[deploy-bot](agent:agent-3)@');
  await evidenceShot(page, 'before-enter-wrong-row.png');
});

test('BEFORE: clicking a row leaves the @query residue with no trailing space', async ({ page }) => {
  const { input, rows } = await openDetail(page);
  await input.fill('@bu');
  await rows.first().click();
  // DEFECT: residue `@bu` glued to the token, no trailing space, caret
  // inside the mess — continued typing breaks the markdown link.
  await expect(input).toHaveValue('[builder](agent:agent-1)@bu');
  await evidenceShot(page, 'before-click-residue.png');
});

test('BEFORE: opening the listbox steals focus into the LAST row (ref defect)', async ({ page }) => {
  const { input, rows } = await openDetail(page);
  await input.fill('@');
  await expect(rows).toHaveCount(3);
  // DEFECT: one ref attached to every row → .current = last row → focus()
  // lands there; the textarea loses focus (typing + IME break).
  const focused = await input.evaluate(() => {
    const el = document.activeElement;
    return { isTextarea: el === document.querySelector('.composer-input'), cls: el?.className ?? '' };
  });
  expect(focused.isTextarea).toBe(false);
  expect(focused.cls).toContain('mention-inline-row');
  await evidenceShot(page, 'before-focus-theft.png');
});
